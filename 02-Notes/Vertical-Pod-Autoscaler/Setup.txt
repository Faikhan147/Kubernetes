Step 4.

# 1️⃣ Ek folder banao
mkdir ~/vpa-tls && cd ~/vpa-tls

# 2️⃣ CA key aur certificate generate karo (valid 1 year)
openssl req -x509 -newkey rsa:4096 -nodes \
  -keyout ca.key -out ca.crt \
  -subj "/CN=VPA-CA" \
  -days 365

# 3️⃣ Server key aur certificate signing request (CSR) banao
openssl req -newkey rsa:4096 -nodes \
  -keyout serverKey.pem -out server.csr \
  -subj "/CN=vpa-admission-controller.kube-system.svc"

# 4️⃣ CSR ko CA se sign karke server certificate banao
openssl x509 -req -in server.csr -CA ca.crt -CAkey ca.key \
  -CAcreateserial -out serverCert.pem -days 365

# 5️⃣ Secret create karo (CA + Server Cert + Server Key)
kubectl create secret generic vpa-tls-certs \
  --from-file=serverKey.pem=./serverKey.pem \
  --from-file=serverCert.pem=./serverCert.pem \
  --from-file=caCert.pem=./ca.crt \
  -n kube-system
